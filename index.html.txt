<script>
(function(){
  let gender='male', goal='کاهش وزن', pace='ملایم', economy='ضعیف', diabetes='ندارم';

  function setupSeg(id, onPick){
    document.getElementById(id).addEventListener('click', (e)=>{
      const btn = e.target.closest('button');
      if(!btn) return;
      [...e.currentTarget.children].forEach(c=>c.classList.remove('active'));
      btn.classList.add('active');
      onPick(btn.dataset.val);
    });
  }
  setupSeg('pfGenderSeg', v=>gender=v);
  setupSeg('pfGoalSeg', v=>{
    goal=v;
    document.getElementById('pfPaceField').style.display = (goal==='حفظ وزن') ? 'none' : 'block';
  });
  setupSeg('pfPaceSeg', v=>pace=v);
  setupSeg('pfEconomySeg', v=>economy=v);
  setupSeg('pfDiabetesSeg', v=>diabetes=v);

  document.getElementById('pfBtn').addEventListener('click', function(){
    const name    = document.getElementById('pfName').value.trim() || '—';
    const contact = document.getElementById('pfContact').value.trim() || '—';
    const age     = document.getElementById('pfAge').value.trim() || '—';
    const height  = document.getElementById('pfHeight').value.trim() || '—';
    const weight  = document.getElementById('pfWeight').value.trim() || '—';
    const activity= document.getElementById('pfActivity').value;
    const notes   = document.getElementById('pfNotes').value.trim();
    const prefs   = [...document.querySelectorAll('#pfPrefs input:checked')].map(el=>el.value);

    let jalali = '';
    try{
      jalali = new Intl.DateTimeFormat('fa-IR-u-ca-persian', {year:'numeric', month:'2-digit', day:'2-digit'}).format(new Date());
    }catch(e){ jalali = ''; }

    let msg = 'درخواست تنظیم رژیم غذایی — دکتر ایرج کوه شکاف\n';
    msg += 'تاریخ: ' + jalali + '\n';
    msg += '——————————————\n';
    msg += 'نام و نام‌خانوادگی: ' + name + '\n';
    msg += 'شماره تماس/آیدی: ' + contact + '\n';
    msg += 'جنسیت: ' + (gender==='male' ? 'مرد' : 'زن') + '\n';
    msg += 'سن: ' + age + ' سال\n';
    msg += 'قد: ' + height + ' سانتی‌متر\n';
    msg += 'وزن: ' + weight + ' کیلوگرم\n';
    msg += 'سطح فعالیت: ' + activity + '\n';
    msg += 'هدف: ' + goal + (goal!=='حفظ وزن' ? (' (سرعت: '+pace+')') : '') + '\n';
    msg += 'وضعیت اقتصادی: ' + economy + '\n';
    msg += 'دیابت: ' + diabetes + '\n';
    msg += 'علایق غذایی: ' + (prefs.length ? prefs.join('، ') : 'ثبت نشده') + '\n';
    if(notes) msg += 'توضیحات/حساسیت‌ها: ' + notes + '\n';

    document.getElementById('pfMsgBox').value = msg;
    document.getElementById('pfOutput').classList.add('show');
    document.getElementById('pfCopied').style.display = 'none';

    try{ document.getElementById('pfOutput').scrollIntoView({behavior:'smooth', block:'start'}); }catch(e){}
  });

  function copyMessage(){
    const box = document.getElementById('pfMsgBox');
    box.select();
    return new Promise((resolve)=>{
      try{
        navigator.clipboard.writeText(box.value).then(()=>{
          document.getElementById('pfCopied').style.display = 'block';
          resolve();
        }).catch(()=>{
          try{ document.execCommand('copy'); }catch(e){}
          document.getElementById('pfCopied').style.display = 'block';
          resolve();
        });
      }catch(e){
        try{ document.execCommand('copy'); }catch(e2){}
        document.getElementById('pfCopied').style.display = 'block';
        resolve();
      }
    });
  }

  document.getElementById('pfCopyBtn').addEventListener('click', function(){
    copyMessage();
  });

  document.getElementById('pfCopyCardBtn').addEventListener('click', function(){
    const cardNumber = document.getElementById('pfCardNumber').textContent.replace(/-/g, '');
    const doneEl = document.getElementById('pfCardCopied');
    try{
      navigator.clipboard.writeText(cardNumber).then(()=>{
        doneEl.style.display = 'block';
      }).catch(()=>{
        const tmp = document.createElement('textarea');
        tmp.value = cardNumber;
        document.body.appendChild(tmp);
        tmp.select();
        try{ document.execCommand('copy'); }catch(e){}
        document.body.removeChild(tmp);
        doneEl.style.display = 'block';
      });
    }catch(e){
      doneEl.style.display = 'block';
    }
  });

  // Eitaa: open the doctor's own Eitaa chat/group directly (copy the text first, so the
  // person can paste it once the chat opens). Rubika/Bale don't have a documented public
  // "prefill message text" link, so we just copy the text and open the app/site.
  const EITAA_DESTINATION = 'https://eitaa.com/joinchat/1716258450C6ae3630200';
  document.getElementById('pfEitaaBtn').addEventListener('click', function(){
    copyMessage().then(()=>{
      window.open(EITAA_DESTINATION, '_blank');
    });
  });
  document.getElementById('pfRubikaBtn').addEventListener('click', function(){
    copyMessage().then(()=>{
      window.open('https://web.rubika.ir/', '_blank');
    });
  });
  document.getElementById('pfBaleBtn').addEventListener('click', function(){
    copyMessage().then(()=>{
      window.open('https://web.bale.ai/', '_blank');
    });
  });
})();
</script>
